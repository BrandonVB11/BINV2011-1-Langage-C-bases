-- Solution du Quizz sur Moodle sur les pointeurs. --

-- 1
Laquelle de ces déclarations d'un pointeur d'entier est correcte?
a.ptr(int) ptrI;

b.int *ptrI;

c.int &ptrI;

-- Solution
La response est B
Correct! L'étoile après le type indique qu'il s'agit d'un pointeur vers ce type.

--2
Après exécution des instructions suivantes, quelle affirmation est correcte?

   int a = 5;
   int* p = &a;
   *p = 10;

a.a vaut 10

b.a vaut 5

c.génrère une SEGFAULT
-- Solution
La response est A

Laquelle de ces allocations de mémoire dynamique est correcte?
Question 3Réponse

a.double *bufferD = (double*)malloc(4*sizeof(double));

b.double *bufferD = (double*)malloc(4);

c.double *bufferD = (double*)malloc( );
--Solution
la response est A car
4*sizeof(double) indique qu'il faut allouer 4 fois de la place pour un double.
sizeof(double) donne le nombre de bytes qu'occupe un double en mémoire.
